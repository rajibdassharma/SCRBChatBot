# ============================================================================
# CyberFraud Data Entry — dependencies for the UPLOAD ANALYSIS jobs only.
#
# Install on the machine that PARSES uploaded files. Since 2026-08-17
# that is the PRODUCTION SERVER — `deploy/install-nightly.sh` installs
# this file into the same venv as the web app. It is no longer
# laptop-only.
#
#     pip install -r requirements.txt -r requirements-analysis.txt
#
# WHY THIS IS STILL A SEPARATE FILE
# ---------------------------------
# The web application never imports the `analysis` package — verified
# across cyber_fraud.py, api/ and models/ — and pdfplumber/xlrd appear
# nowhere outside analysis/. Nothing a user request touches needs them.
#
# The separation is risk containment, and it matters MORE now that both
# sets live in one venv. pdfplumber 0.11.10+ requires Pillow>=12.2, and
# Pillow is what reportlab renders every operator-facing PDF with —
# case files, mule reports, the daily DSR sheets. An unpinned
# `pip install pdfplumber` would upgrade Pillow underneath reportlab for
# ~90 users.
#
# The pins below are what prevent that. Do not relax them without
# rendering a case-file PDF and a mule report and looking at the output;
# install-nightly.sh imports both packages after installing and prints
# their versions for exactly this reason.
#
# HISTORY: results used to be exported from the laptop and imported into
# production as a SQL file. That path is retired — `import-analysis.sh`
# survives for restoring an old export, and the matching export script
# is gone.
# ============================================================================

# PDF statement reading. 0.11.4 pinned deliberately: later releases
# require Pillow>=12.2, which conflicts with other tooling commonly
# present on a shared development machine (streamlit pins pillow<12).
# 0.11.4 is the version the current corpus was parsed with.
pdfplumber==0.11.4

# Legacy .xls statements. 2.0 dropped .xlsx support on purpose —
# openpyxl (in requirements.txt) covers those.
xlrd>=2.0.1

# ID-photo decoding for the duplicate-ID hashes. Arrives transitively
# via reportlab, but analysis/hash_id_photos.py imports PIL directly, so
# it is declared rather than assumed. Capped below 12 to match the
# pdfplumber pin above.
pillow>=10.0.0,<12
